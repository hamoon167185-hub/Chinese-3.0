function initApp() {
  // Toàn bộ logic ứng dụng cũ được chạy ở đây sau khi nhập đúng mật khẩu
  let levels = [...new Set(VOCAB_DATA.map(x => x[0]))].sort((a,b)=>a-b);
  let activeLevels = new Set(levels);
  let currentPool = [];
  let currentIndex = 0;
  let knownSet = new Set();
  let unknownSet = new Set();
  let orderIndices = [];
  let isFlipped = false;

  const levelChips = document.getElementById('levelChips');
  levels.forEach(lvl => {
    let count = VOCAB_DATA.filter(x => x[0] === lvl).length;
    let label = document.createElement('label');
    label.className = 'chip';
    label.innerHTML = `<input type="checkbox" value="${lvl}" checked> HSK ${lvl} <span class="chip-count">(${count})</span>`;
    label.querySelector('input').addEventListener('change', (e) => {
      if(e.target.checked) activeLevels.add(lvl); else activeLevels.delete(lvl);
      rebuildPool();
    });
    levelChips.appendChild(label);
  });

  function rebuildPool() {
    currentPool = VOCAB_DATA.filter(x => activeLevels.has(x[0]));
    document.getElementById('poolCount').textContent = currentPool.length;
    orderIndices = currentPool.map((_, i) => i);
    currentIndex = 0;
    knownSet.clear();
    unknownSet.clear();
    updateStats();
    renderCard();
  }

  document.getElementById('selectAllBtn').addEventListener('click', () => {
    document.querySelectorAll('#levelChips input').forEach(el => el.checked = true);
    activeLevels = new Set(levels);
    rebuildPool();
  });
  document.getElementById('selectNoneBtn').addEventListener('click', () => {
    document.querySelectorAll('#levelChips input').forEach(el => el.checked = false);
    activeLevels.clear();
    rebuildPool();
  });

  const cardFace = document.getElementById('cardFace');
  const levelTag = document.getElementById('levelTag');
  const hanziDisplay = document.getElementById('hanziDisplay');
  const pinyinDisplay = document.getElementById('pinyinDisplay');
  const posDisplay = document.getElementById('posDisplay');
  const meaningDisplay = document.getElementById('meaningDisplay');
  const cardPos = document.getElementById('cardPos');
  const fcEmpty = document.getElementById('fcEmpty');
  const cardWrap = document.getElementById('cardWrap');

  function renderCard() {
    if(currentPool.length === 0) {
      fcEmpty.style.display = 'block';
      cardWrap.style.display = 'none';
      return;
    }
    fcEmpty.style.display = 'none';
    cardWrap.style.display = 'flex';

    if(currentIndex < 0) currentIndex = currentPool.length - 1;
    if(currentIndex >= currentPool.length) currentIndex = 0;

    let realIdx = orderIndices[currentIndex];
    let item = currentPool[realIdx];

    levelTag.textContent = `HSK ${item[0]}`;
    hanziDisplay.textContent = item[1];
    pinyinDisplay.textContent = item[2];
    posDisplay.textContent = item[3] || '';
    meaningDisplay.textContent = item[4];

    isFlipped = false;
    cardFace.classList.remove('flipped');
    cardPos.textContent = `${currentIndex + 1} / ${currentPool.length}`;
  }

  function toggleFlip() {
    isFlipped = !isFlipped;
    cardFace.classList.toggle('flipped', isFlipped);
  }

  cardFace.addEventListener('click', toggleFlip);
  document.getElementById('flipBtn').addEventListener('click', toggleFlip);

  document.getElementById('nextBtn').addEventListener('click', () => { currentIndex++; renderCard(); });
  document.getElementById('prevBtn').addEventListener('click', () => { currentIndex--; renderCard(); });

  document.addEventListener('keydown', (e) => {
    if(document.getElementById('mainAppContainer').style.display !== 'block') return;
    if(e.code === 'Space') { e.preventDefault(); toggleFlip(); }
    else if(e.code === 'ArrowRight') { currentIndex++; renderCard(); }
    else if(e.code === 'ArrowLeft') { currentIndex--; renderCard(); }
  });

  document.getElementById('shuffleBtn').addEventListener('click', () => {
    for (let i = orderIndices.length - 1; i > 0; i--) {
      let j = Math.floor(Math.random() * (i + 1));
      [orderIndices[i], orderIndices[j]] = [orderIndices[j], orderIndices[i]];
    }
    currentIndex = 0;
    renderCard();
  });

  document.getElementById('resetOrderBtn').addEventListener('click', () => {
    orderIndices = currentPool.map((_, i) => i);
    currentIndex = 0;
    renderCard();
  });

  function updateStats() {
    document.getElementById('knownCount').textContent = knownSet.size;
    document.getElementById('unknownCount').textContent = unknownSet.size;
  }

  document.getElementById('knowBtn').addEventListener('click', () => {
    let realIdx = orderIndices[currentIndex];
    knownSet.add(realIdx);
    unknownSet.delete(realIdx);
    updateStats();
    currentIndex++;
    renderCard();
  });

  document.getElementById('dontKnowBtn').addEventListener('click', () => {
    let realIdx = orderIndices[currentIndex];
    unknownSet.add(realIdx);
    knownSet.delete(realIdx);
    updateStats();
    currentIndex++;
    renderCard();
  });

  // Phát âm chuẩn Web Speech API
  function speakCurrent() {
    if(currentPool.length === 0) return;
    let item = currentPool[orderIndices[currentIndex]];
    if('speechSynthesis' in window) {
      window.speechSynthesis.cancel();
      let utter = new SpeechSynthesisUtterance(item[1]);
      utter.lang = 'zh-CN';
      utter.rate = 0.9;
      window.speechSynthesis.speak(utter);
    }
  }
  document.getElementById('audioBtn').addEventListener('click', speakCurrent);

  // Xem nét chữ HanziWriter
  document.getElementById('strokeBtn').addEventListener('click', () => {
    if(currentPool.length === 0) return;
    let item = currentPool[orderIndices[currentIndex]];
    let hanziStr = item[1];
    document.getElementById('strokeWordTitle').textContent = `${hanziStr} (${item[2]})`;
    let container = document.getElementById('strokeChars');
    container.innerHTML = '';
    
    for(let char of hanziStr) {
      if(/[\u4e00-\u9fa5]/.test(char)) {
        let box = document.createElement('div');
        box.className = 'stroke-char-box';
        box.innerHTML = `<div class="stroke-target" id="target_${Math.random().toString(36).substring(2,7)}"></div>
                         <div class="stroke-char-label">${char}</div>
                         <div><button class="mini-btn animate-btn">Viết lại</button> <button class="mini-btn quiz-btn">Tập viết</button></div>`;
        container.appendChild(box);
        let targetId = box.querySelector('.stroke-target').id;
        try {
          let writer = HanziWriter.create(targetId, char, { width: 160, height: 160, padding: 5, strokeColor: '#2a2013' });
          box.querySelector('.animate-btn').addEventListener('click', () => writer.animateCharacter());
          box.querySelector('.quiz-btn').addEventListener('click', () => writer.quiz());
        } catch(err) {}
      }
    }
    document.getElementById('strokeOverlay').classList.add('open');
  });

  document.getElementById('strokeClose').addEventListener('click', () => {
    document.getElementById('strokeOverlay').classList.remove('open');
  });

  // Tabs navigation
  document.querySelectorAll('.tabs .tab-btn').forEach(btn => {
    btn.addEventListener('click', (e) => {
      document.querySelectorAll('.tabs .tab-btn').forEach(b => b.classList.remove('active'));
      document.querySelectorAll('.panel').forEach(p => p.classList.remove('active'));
      e.target.classList.add('active');
      document.getElementById(e.target.dataset.panel).classList.add('active');
    });
  });

  // Tìm kiếm nhanh
  const searchInput = document.getElementById('searchInput');
  const searchResults = document.getElementById('searchResults');
  searchInput.addEventListener('input', (e) => {
    let keyword = e.target.value.trim().toLowerCase();
    if(!keyword) { searchResults.style.display = 'none'; return; }
    let matches = VOCAB_DATA.filter(x => x[1].toLowerCase().includes(keyword) || x[2].toLowerCase().includes(keyword) || x[4].toLowerCase().includes(keyword)).slice(0, 15);
    if(matches.length === 0) {
      searchResults.innerHTML = `<div style="padding:10px 14px; color:var(--muted); font-size:13px;">Không tìm thấy từ phù hợp</div>`;
    } else {
      searchResults.innerHTML = matches.map(m => `
        <div class="search-item" data-hanzi="${m[1]}">
          <span class="sr-hanzi">${m[1]}</span>
          <span class="sr-pinyin">${m[2]}</span>
          <span class="sr-meaning">${m[4]}</span>
        </div>
      `).join('');
      searchResults.querySelectorAll('.search-item').forEach(el => {
        el.addEventListener('click', () => {
          let h = el.dataset.hanzi;
          let idx = currentPool.findIndex(x => x[1] === h);
          if(idx !== -1) {
            currentIndex = idx;
            renderCard();
            document.querySelector('[data-panel="panel-flashcard"]').click();
          }
          searchResults.style.display = 'none';
          searchInput.value = '';
        });
      });
    }
    searchResults.style.display = 'block';
  });
  document.addEventListener('click', (e) => { if(!e.target.closest('.search-wrap')) searchResults.style.display = 'none'; });

  // Khởi chạy ban đầu
  rebuildPool();
}
</script>
